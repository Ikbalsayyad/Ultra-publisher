import time, machine
soil = machine.ADC(machine.Pin(34))
pump = machine.Pin(5, machine.Pin.OUT)

while True:
    if (soil.read()/4095)*100 < 30:
        pump.value(1) # Pump ON
    else: pump.value(0)
    time.sleep(5)
